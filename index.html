구글 스프레드시트(Apps Script) 연동

구글 스프레드시트와 실시간 연동하여 학생 명단을 가져오고, 배정 결과를 시트에 즉시 저장할 수 있습니다.

원클릭 스프레드시트 붙여넣기 (TSV) DIRECT CLIPBOARD

구글 스프레드시트 임의의 셀을 클릭한 뒤 Ctrl + V를 누르면 바로 표로 입력됩니다.

* 저장된 URL은 향후 상단 [SHEET LOAD] 및 [SHEET SAVE] 클릭 시 즉시 비동기 통신에 사용됩니다.

// [구글 스프레드시트 > 확장 프로그램 > Apps Script]에 붙여넣고

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var students = [];
  var roles = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) students.push(data[i][0]);
    if (data[i][1]) {
      roles.push({
        name: data[i][1],
        desc: data[i][2] || "성실히 직무를 실천합니다"
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({
    students: students,
    roles: roles
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var payload = JSON.parse(e.postData.contents);
  var assignments = payload.assignments;
  
  sheet.clear();
  sheet.appendRow(["번호", "학생 이름", "배정된 역할", "역할 임무", "배정일시"]);
  
  for (var i = 0; i < assignments.length; i++) {
    var item = assignments[i];
    sheet.appendRow([i + 1, item.student, item.role, item.desc, new Date()]);
  }
  
  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    count: assignments.length
  })).setMimeType(ContentService.MimeType.JSON);
}
