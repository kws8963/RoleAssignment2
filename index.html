구글 스프레드시트(Apps Script) 연동

구글 스프레드시트와 연결하여 학생 명단과 역할을 손쉽게 가져오고, 배정 결과를 바로 시트에 저장할 수 있습니다.

⚡ 지금 바로 시트에 붙여넣기 (원클릭) 간편 기능

스프레드시트에 바로 Ctrl + V로 붙여넣을 수 있는 표 형태로 복사됩니다.

💡 주소를 저장해두면 상단의 [🔄 시트 불러오기] 및 [💾 시트에 결과 저장] 클릭 시 실시간 API 통신이 작동합니다.

📜 Google Apps Script 배포 코드 템플릿 보기 (복사 가능) 
// [구글 스프레드시트 > 확장 프로그램 > Apps Script]에 붙여넣고
// [배포 > 새 배포 > 웹 앱 (액세스 권한: 모든 사용자)]로 배포하세요.

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  // 1행 헤더 제외, A열: 학생이름, B열: 역할이름, C열: 역할설명
  var students = [];
  var roles = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) students.push(data[i][0]);
    if (data[i][1]) {
      roles.push({
        name: data[i][1],
        desc: data[i][2] || "즐겁게 성실히 실천해요"
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({
    students: students,
    roles: roles
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var payload = JSON.parse(e.postData.contents);
  var assignments = payload.assignments; // [{student, role, desc}]
  
  sheet.clear();
  sheet.appendRow(["번호", "학생 이름", "배정된 역할", "역할 설명", "배정일시"]);
  
  for (var i = 0; i < assignments.length; i++) {
    var item = assignments[i];
    sheet.appendRow([i + 1, item.student, item.role, item.desc, new Date()]);
  }
  
  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    count: assignments.length
  })).setMimeType(ContentService.MimeType.JSON);
}
