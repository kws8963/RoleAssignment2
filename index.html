구글 스프레드시트 웹앱(Apps Script) 연동 설정

구글 스프레드시트의 Apps Script 웹앱 URL을 등록하면, 학생 명단 및 역할을 실시간으로 불러오고 배정 결과를 시트에 즉시 영구 저장할 수 있습니다.

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var students = [];
  var roles = [];
  var results = [];
  
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) students.push(data[i][0]);
    if (data[i][1]) {
      roles.push({
        name: data[i][1],
        desc: data[i][2] || "성실히 직무를 실천합니다"
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    students: students,
    roles: roles,
    results: results
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var payload = JSON.parse(e.postData.contents);
  var assignments = payload.assignments || [];
  
  sheet.clear();
  sheet.appendRow(["번호", "학생 이름", "배정된 역할", "역할 임무", "배정일시"]);
  
  for (var i = 0; i < assignments.length; i++) {
    var item = assignments[i];
    sheet.appendRow([i + 1, item.student, item.role, item.desc, new Date()]);
  }
  
  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    count: assignments.length
  })).setMimeType(ContentService.MimeType.JSON);
}
